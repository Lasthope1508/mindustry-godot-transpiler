def read(p,base=ROOT):return (base/p).read_text(encoding='utf-8')

def merge(a,b,c):
 if a==b:return copy.deepcopy(c)
 if isinstance(a,dict) and isinstance(b,dict) and isinstance(c,dict):
  d=copy.deepcopy(c)
  for k in dict.fromkeys([*a,*b]):
   if k not in b:assert d[k]==a[k];del d[k]
   elif k not in a:assert k not in d or d[k]==b[k];d[k]=copy.deepcopy(b[k])
   elif a[k]!=b[k]:d[k]=merge(a[k],b[k],d[k])
  return d
 if isinstance(a,list) and isinstance(b,list) and isinstance(c,list):
  if all(isinstance(x,dict) and 'id' in x for x in a+b+c):
   am,bm,cm=({x['id']:x for x in values} for values in [a,b,c]);assert set(am)<=set(bm) and set(am)==set(cm)
   return [merge(am[x['id']],x,cm[x['id']]) if x['id'] in am else copy.deepcopy(x) for x in b]
  assert len(a)==len(b)==len(c)
  return [merge(x,y,z) for x,y,z in zip(a,b,c)]
 assert c==a or c==b
 return copy.deepcopy(b)

def delta(a,b,c):
 x,y=a.splitlines(keepends=True),b.splitlines(keepends=True)
 for group in reversed(list(difflib.SequenceMatcher(None,x,y,autojunk=False).get_grouped_opcodes(1))):
  old=''.join(x[group[0][1]:group[-1][2]]);new=''.join(y[group[0][3]:group[-1][4]])
  if c.count(old)==1:c=c.replace(old,new,1);continue
  for tag,i,j,p,q in reversed(group):
   if tag=='equal':continue
   old,new=''.join(x[i:j]),''.join(y[p:q]);assert old and c.count(old)==1,old;c=c.replace(old,new,1)
 return c

def content_delta(a,b):
 assert {k:v for k,v in a.items() if k not in ('blocks','counts')}=={k:v for k,v in b.items() if k not in ('blocks','counts')}
 assert {k:v for k,v in a['counts'].items() if k!='blocks'}=={k:v for k,v in b['counts'].items() if k!='blocks'}
 am,bm=({x['id']:x for x in data['blocks']} for data in [a,b]);assert set(bm)-set(am)==ADDED and set(am)<=set(bm)
 assert b['counts']['blocks']==a['counts']['blocks']+len(ADDED)==len(b['blocks'])
 changed=[]
 for name,old in am.items():
  new=bm[name]
  assert {k:v for k,v in old.items() if k!='properties'}=={k:v for k,v in new.items() if k!='properties'}
  x,y=old['properties'],new['properties'];assert all(y.get(k)==v for k,v in x.items());assert set(y)-set(x)<= {'itemDrop','cacheLayer'}
  if old!=new:changed.append(name)
 assert [x['id'] for x in b['blocks'] if x['id'] in am]==[x['id'] for x in a['blocks']]
 return {'added_blocks':sorted(ADDED),'changed_existing_blocks':changed,'counts_before':a['counts'],'counts_after':b['counts']}
