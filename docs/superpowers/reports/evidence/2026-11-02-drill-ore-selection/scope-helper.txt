def merge(a,b,c):
 if a==b:return copy.deepcopy(c)
 if isinstance(a,dict) and isinstance(b,dict) and isinstance(c,dict):
  d=copy.deepcopy(c)
  for k in dict.fromkeys([*a,*b]):
   if k not in b:assert d[k]==a[k];del d[k]
   elif k not in a:assert k not in d or d[k]==b[k];d[k]=copy.deepcopy(b[k])
   elif a[k]!=b[k]:d[k]=merge(a[k],b[k],d[k])
  return d
 assert c==a or c==b
 return copy.deepcopy(b)
