import json
import re
import subprocess
from pathlib import Path

ROOT = Path(r'C:/Users/Admin/Documents/Game/mindustry-godot-transpiler-copy-20260714-073528')
def git(*args):
    return subprocess.check_output(['git',*args],cwd=ROOT).decode('utf-8')
assert git('rev-parse','HEAD').startswith('05fd5d87')
paths = git('diff','--cached','--name-only').splitlines()
assert not any(p.endswith('.gd.uid') or p.startswith('arc-upstream/') or p.endswith('/project.godot') or p.endswith('/source_ssot_modules.json') for p in paths)
expected = {'source_symbol_registry.json':('symbols',17), 'source_api_contracts.json':('contracts',2), 'source_state_schema.json':('schemas',6), 'source_slice_ledger.json':('slices',2)}
out = []
for name,(key,count) in expected.items():
    path = 'godot-mindustry/src/generated/' + name
    old,new = json.loads(git('show','HEAD:'+path)), json.loads(git('show',':'+path))
    assert all(new[key].get(k)==v for k,v in old[key].items())
    assert len(set(new[key])-set(old[key]))==count
    assert {k:v for k,v in old.items() if k!=key} == {k:v for k,v in new.items() if k!=key}
    out.append(name+': only '+str(count)+' new entries')
assert re.search(r'^\d+ passed, \d+ subtests passed in ',Path('C:/tmp/liquid-consumer-full.log').read_text(encoding='utf-8'),re.M)
assert Path('C:/tmp/liquid-consumer-native-accepted').exists()
assert json.loads(Path('C:/tmp/liquid-consumer-replay.json').read_text(encoding='utf-8'))['matches']
assert not json.loads(Path('C:/tmp/liquid-consumer-validator-final.json').read_text(encoding='utf-8'))['errors']
subprocess.run(['git','diff','--cached','--check'],cwd=ROOT,check=True)
out += ['full/native/editor/replay/catalog gates accepted','no unrelated project/uid/arc/manifest staged','index diff check PASS','staged files before audit evidence: '+str(len(paths))]
path = ROOT / 'docs/superpowers/reports/evidence/2026-11-01-liquid-consumer-hydration/index-final.log'
path.write_text('\n'.join(out)+'\n',encoding='utf-8')
print('\n'.join(out))
