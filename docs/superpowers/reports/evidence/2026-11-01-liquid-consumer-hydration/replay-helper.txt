import hashlib
import importlib.util
import json
import subprocess
import sys
from pathlib import Path

ROOT = Path(r'C:/Users/Admin/Documents/Game/mindustry-godot-transpiler-copy-20260714-073528')
spec = importlib.util.spec_from_file_location('scope','C:/tmp/liquid-consumer-scope.py')
scope = importlib.util.module_from_spec(spec)
spec.loader.exec_module(scope)
paths = scope.JSON_PATHS + [scope.GEN + n for n in ['content_registry.json','MindustryContentRegistry.gd','source_ssot_modules.json','source_symbol_audit_report.json']]
def hashes():
    return {p:hashlib.sha256((ROOT/p).read_bytes()).hexdigest() for p in paths}
before = hashes()
subprocess.run([sys.executable,'tools/mindustry_godot_transpiler/transpile.py','content'],cwd=ROOT,check=True)
subprocess.run([sys.executable,'tools/mindustry_godot_transpiler/register_liquid_consumer_hydration.py'],cwd=ROOT,check=True)
after = hashes()
report = {'matches': before == after, 'files': {p:{'before':before[p],'after':after[p],'matches':before[p]==after[p]} for p in paths}}
Path('C:/tmp/liquid-consumer-replay.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
assert report['matches'], [p for p in paths if before[p]!=after[p]]
print('content + registration replay unchanged:',len(paths),'files')
